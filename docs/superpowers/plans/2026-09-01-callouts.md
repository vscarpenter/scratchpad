**Goal:** Ship `> [!NOTE]` callouts
(`docs/superpowers/specs/2026-09-01-callouts-design.md`) as v3.23.0.

**Architecture:** `public/js/markdown.js` gains a `walkTokens` hook that
tags blockquote tokens whose first inline text starts with a marker, and a
`blockquote` renderer that emits the callout wrapper; `decorateRendered`
skips callouts. `public/css/app.css` gains token-only callout rules and
dual-scopes the quote and code rules to `.share-body`.

**Tech Stack:** marked 18 (`walkTokens`, renderer overrides), DOMPurify,
Playwright on three browsers.

**Spec:** `docs/superpowers/specs/2026-09-01-callouts-design.md`

- [ ] **Step 3: Renderer.** In `markdown.js`, after the checkbox `marked.use`
  block, add:

```js
  const CALLOUT_PATTERN = /^\[!(note|tip|important|warning|caution)\](?:[ \t]+([^\n]*))?(?:\n|$)/i;

  // GitHub-style callouts: a blockquote whose first inline text starts with
  // [!KIND]. The marker line is stripped from the token before rendering.
  function tagCallout(token) {
    if (token.type !== 'blockquote' || !token.tokens || !token.tokens.length) return;
    const paragraph = token.tokens[0];
    const inline = paragraph.type === 'paragraph' && paragraph.tokens ? paragraph.tokens[0] : null;
    if (!inline || inline.type !== 'text') return;
    const match = CALLOUT_PATTERN.exec(inline.text);
    if (!match) return;
    token.callout = { kind: match[1].toLowerCase(), title: (match[2] || '').trim() };
    inline.text = inline.text.slice(match[0].length);
    inline.raw = inline.text;
    if (!inline.text && paragraph.tokens.length === 1) token.tokens.shift();
  }

  function renderBlockquote(token) {
    const body = this.parser.parse(token.tokens);
    if (!token.callout) return '<blockquote>\n' + body + '</blockquote>\n';
    const kind = token.callout.kind;
    const title = token.callout.title || kind.charAt(0).toUpperCase() + kind.slice(1);
    return (
      '<blockquote class="callout callout-' + kind + '"><p class="callout-title">' + escapeHtml(title) + '</p>' +
      body +
      '</blockquote>\n'
    );
  }

  if (window.marked && typeof window.marked.use === 'function') {
    window.marked.use({ walkTokens: tagCallout, renderer: { blockquote: renderBlockquote } });
  }
```

`escapeHtml` is defined later in the file as a function declaration, so it
is hoisted. In `decorateRendered`, add
`if (bq.classList.contains('callout')) continue;` as the first line of the
blockquote loop.
